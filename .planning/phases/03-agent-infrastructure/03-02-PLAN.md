# Plan 03-02: Agent 14 — capital-healthcheck

## Goal

Deploy Agent 14 (`capital-healthcheck`) as the platform health monitor. This agent runs every 15 minutes and monitors all other agents, Supabase, Stripe webhooks, email queue, and pending signatures. It must be deployed BEFORE other agents because it monitors them.

## Context

- Schedule: Every 15 minutes (node-cron: `'*/15 * * * *'`)
- Daily 5am UTC report to admin Telegram
- Logs all health checks to `platform_health_log`
- Auto-fixes silently when possible; only alerts if fix fails or critical
- No public Railway domain — runs as standalone Node.js process

## Security Requirements

- Uses `agents/lib/supabase.js` (service_role key)
- All Telegram alerts via `agents/lib/telegram.js` (HMAC-signed)
- Never logs raw API keys or secrets — uses `agents/lib/logger.js`
- Checks `SUPABASE_SERVICE_KEY` is present and is service_role (not anon)
- No external HTTP calls except to verified Telegram endpoint and health check targets

### Task 2: agents/capital-healthcheck/checks/supabase.js

Check Supabase connectivity and RLS compliance:

```javascript
'use strict'
const { getServiceClient } = require('../../lib/supabase')

async function checkSupabase() {
  const supabase = getServiceClient()
  const results = []

  // Check DB connectivity
  const { error: pingErr } = await supabase.from('platform_settings').select('key').limit(1)
  results.push({
    check: 'supabase_connectivity',
    status: pingErr ? 'fail' : 'ok',
    error: pingErr?.message,
  })

  // Check for tables missing RLS (critical — S11-012 enforcement)
  const { data: noRls } = await supabase.rpc('get_tables_without_rls').catch(() => ({ data: null }))
  if (noRls && noRls.length > 0) {
    results.push({
      check: 'rls_audit',
      status: 'critical',
      error: `Tables missing RLS: ${noRls.join(', ')}`,
    })
  } else {
    results.push({ check: 'rls_audit', status: 'ok' })
  }

  // Check pending signatures older than 5 days
  const fiveDaysAgo = new Date(Date.now() - 5 * 24 * 60 * 60 * 1000).toISOString()
  const { count: staleSigs } = await supabase
    .from('signature_requests')
    .select('*', { count: 'exact', head: true })
    .eq('status', 'pending')
    .lt('created_at', fiveDaysAgo)
    .catch(() => ({ count: 0 }))

  if (staleSigs > 0) {
    results.push({
      check: 'stale_signatures',
      status: 'warn',
      error: `${staleSigs} pending signatures older than 5 days`,
    })
  }

  // Check fee_invoices in suspended state with no admin review
  const { count: suspendedFees } = await supabase
    .from('fee_invoices')
    .select('*', { count: 'exact', head: true })
    .eq('service_suspended', true)
    .eq('paid', false)
    .catch(() => ({ count: 0 }))

  if (suspendedFees > 0) {
    results.push({
      check: 'suspended_invoices',
      status: 'warn',
      error: `${suspendedFees} fee invoices in suspended state need admin review`,
    })
  }

  return results
}

module.exports = { checkSupabase }
```

### Task 3: agents/capital-healthcheck/checks/services.js

Check external service health (Anthropic API, email queue):

'use strict'
const https = require('https')
const { getServiceClient } = require('../../lib/supabase')

async function checkAnthropicApi() {
  return new Promise((resolve) => {
    const req = https.request(
      'https://api.anthropic.com/v1/messages',
      { method: 'POST', headers: {
        'anthropic-version': '2023-06-01',
        'x-api-key': process.env.ANTHROPIC_API_KEY || '',
        'content-type': 'application/json',
      }},
      (res) => {
        // 400 = valid auth, bad request = API is UP
        // 401 = API key issue
        if (res.statusCode === 400) resolve({ check: 'anthropic_api', status: 'ok' })
        else if (res.statusCode === 401) resolve({ check: 'anthropic_api', status: 'fail', error: 'Invalid API key' })
        else resolve({ check: 'anthropic_api', status: 'warn', error: `Unexpected status ${res.statusCode}` })
      }
    )
    req.on('error', (e) => resolve({ check: 'anthropic_api', status: 'fail', error: e.message }))
    req.write(JSON.stringify({ model: 'claude-sonnet-4-6', max_tokens: 1, messages: [] }))
    req.end()
  })
}

async function checkEmailQueue() {
  const supabase = getServiceClient()
  
  // Check for stuck emails (sent=false, not requiring approval, older than 1 hour)
  const oneHourAgo = new Date(Date.now() - 60 * 60 * 1000).toISOString()
  const { count: stuckEmails } = await supabase
    .from('agent_emails')
    .select('*', { count: 'exact', head: true })
    .eq('sent', false)
    .eq('requires_client_approval', false)
    .eq('requires_admin_approval', false)
    .lt('created_at', oneHourAgo)
    .catch(() => ({ count: 0 }))

  return {
    check: 'email_queue',
    status: stuckEmails > 0 ? 'warn' : 'ok',
    error: stuckEmails > 0 ? `${stuckEmails} emails stuck in queue` : undefined,
  }
}

async function checkEnvVars() {
  const required = [
    'SUPABASE_URL', 'SUPABASE_SERVICE_KEY', 'ANTHROPIC_API_KEY',
    'RESEND_API_KEY', 'TELEGRAM_BOT_TOKEN', 'TELEGRAM_CHAT_ID',
    'TELEGRAM_SIGNING_SECRET', 'INTERNAL_SERVICE_SECRET',
  ]
  const missing = required.filter((k) => !process.env[k])
  return {
    check: 'env_vars',
    status: missing.length > 0 ? 'fail' : 'ok',
    error: missing.length > 0 ? `Missing: ${missing.join(', ')}` : undefined,
  }
}

module.exports = { checkAnthropicApi, checkEmailQueue, checkEnvVars }

### Task 4: agents/capital-healthcheck/checks/agents.js

Check other agents' last run via agent_logs:

```javascript
'use strict'
const { getServiceClient } = require('../../lib/supabase')

// Agents with their expected max interval between runs (ms)
const AGENT_SCHEDULES = {
  'capital-grant-finder': 7 * 24 * 60 * 60 * 1000,    // weekly
  'capital-lender-matcher': 7 * 24 * 60 * 60 * 1000,   // weekly
  'capital-credit-coach': 7 * 24 * 60 * 60 * 1000,     // weekly (on signup)
}

async function checkAgentHealth() {
  const supabase = getServiceClient()
  const results = []

  for (const [agentName, maxIntervalMs] of Object.entries(AGENT_SCHEDULES)) {
    const { data: lastRun } = await supabase
      .from('agent_logs')
      .select('created_at, error')
      .eq('agent_name', agentName)
      .order('created_at', { ascending: false })
      .limit(1)
      .single()
      .catch(() => ({ data: null }))

    if (!lastRun) {
      // Agent has never run — only warn if it's been > 24 hours since deploy
      results.push({ check: `agent_${agentName}`, status: 'warn', error: 'Never run' })
      continue
    }

    const lastRunAge = Date.now() - new Date(lastRun.created_at).getTime()
    
    if (lastRun.error) {
      results.push({ check: `agent_${agentName}`, status: 'warn', error: `Last run had error: ${lastRun.error}` })
    } else if (lastRunAge > maxIntervalMs * 1.5) {
      results.push({ check: `agent_${agentName}`, status: 'warn', error: `Overdue — last run ${Math.round(lastRunAge / 3600000)}h ago` })
    } else {
      results.push({ check: `agent_${agentName}`, status: 'ok' })
    }
  }

  return results
}

module.exports = { checkAgentHealth }
```

### Task 5: agents/capital-healthcheck/index.js (main entry)

```javascript
'use strict'
require('dotenv').config()
const cron = require('node-cron')
const { getServiceClient } = require('../lib/supabase')
const { sendAlert } = require('../lib/telegram')
const logger = require('../lib/logger')
const { checkSupabase } = require('./checks/supabase')
const { checkAnthropicApi, checkEmailQueue, checkEnvVars } = require('./checks/services')
const { checkAgentHealth } = require('./checks/agents')

const AGENT_NAME = 'capital-healthcheck'

async function runHealthCheck() {
  const supabase = getServiceClient()
  const startMs = Date.now()
  logger.info(AGENT_NAME, 'Health check started')

  let allResults = []

  try {
    const [supabaseResults, anthropic, emailQueue, envVars, agentResults] = await Promise.allSettled([
      checkSupabase(),
      checkAnthropicApi(),
      checkEmailQueue(),
      checkEnvVars(),
      checkAgentHealth(),
    ])

    const collect = (r) => Array.isArray(r.value) ? r.value : [r.value || { check: 'unknown', status: r.status === 'fulfilled' ? 'ok' : 'fail', error: r.reason?.message }]
    allResults = [
      ...collect(supabaseResults),
      collect(anthropic)[0],
      collect(emailQueue)[0],
      collect(envVars)[0],
      ...collect(agentResults),
    ]
  } catch (err) {
    logger.error(AGENT_NAME, 'Health check run failed', { error: err.message })
    allResults = [{ check: 'run_error', status: 'fail', error: err.message }]
  }

  // Write all results to platform_health_log
  const inserts = allResults.map((r) => ({
    check_type: r.check,
    service_name: r.check,
    status: r.status,
    error_details: r.error || null,
    alert_sent: false,
  }))

  await supabase.from('platform_health_log').insert(inserts).catch((e) =>
    logger.error(AGENT_NAME, 'Failed to write health log', { error: e.message })
  )

  // Alert on critical/fail
  const critical = allResults.filter((r) => r.status === 'fail' || r.status === 'critical')
  if (critical.length > 0) {
    const msg = `🚨 <b>Cosby Capital Health Alert</b>\n${critical.map((r) => `• ${r.check}: ${r.error}`).join('\n')}`
    await sendAlert(msg, 'error').catch((e) => logger.error(AGENT_NAME, 'Telegram alert failed', { error: e.message }))
    
    await supabase.from('platform_health_log')
      .update({ alert_sent: true })
      .in('check_type', critical.map((r) => r.check))
      .order('created_at', { ascending: false })
  }

  // Log to agent_logs
  await supabase.from('agent_logs').insert({
    agent_name: AGENT_NAME,
    action: 'health_check',
    result: JSON.stringify({ total: allResults.length, ok: allResults.filter((r) => r.status === 'ok').length, issues: critical.length }),
    duration_ms: Date.now() - startMs,
    error: critical.length > 0 ? `${critical.length} issues found` : null,
  })

  logger.info(AGENT_NAME, 'Health check complete', { total: allResults.length, issues: critical.length })
}

async function runDailyReport() {
  const supabase = getServiceClient()
  logger.info(AGENT_NAME, 'Daily report started')

  const yesterday = new Date(Date.now() - 24 * 60 * 60 * 1000).toISOString()
  const { data: issues } = await supabase
    .from('platform_health_log')
    .select('check_type, status, error_details, created_at')
    .neq('status', 'ok')
    .gte('created_at', yesterday)
    .order('created_at', { ascending: false })
    .limit(20)
    .catch(() => ({ data: [] }))

  const { count: totalClients } = await supabase
    .from('clients')
    .select('*', { count: 'exact', head: true })
    .catch(() => ({ count: '?' }))

  const msg = `📊 <b>Cosby Capital Daily Health Report</b>\n${new Date().toUTCString()}\n\nClients: ${totalClients}\nIssues (24h): ${issues?.length || 0}\n${issues?.map((i) => `• ${i.check_type}: ${i.error_details || i.status}`).join('\n') || 'All systems OK ✅'}`
  await sendAlert(msg, 'info').catch((e) => logger.error(AGENT_NAME, 'Daily report alert failed', { error: e.message }))
}

// Every 15 minutes
cron.schedule('*/15 * * * *', runHealthCheck)

// Daily 5am UTC
cron.schedule('0 5 * * *', runDailyReport)

// Run immediately on startup
runHealthCheck().catch((e) => logger.error(AGENT_NAME, 'Startup health check failed', { error: e.message }))

logger.info(AGENT_NAME, 'capital-healthcheck started', { schedule: '*/15 * * * *' })
```
