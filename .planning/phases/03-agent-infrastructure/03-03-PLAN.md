# Plan 03-03: Agent 1 — capital-grant-finder

## Goal

Build and deploy the `capital-grant-finder` agent. This agent scores all active grants against each client's profile using the existing `score_grant_match()` DB function, generates match reasons via Claude, and emails clients when their matches are ready. It runs on new client signup (via `agent_trigger_queue`) and weekly on Mondays at 7am UTC.

## Context

- Schedule: Monday 7am UTC + immediately on new client signup
- Trigger mechanism: polls `agent_trigger_queue` table every 5 minutes; also has Monday cron
- Inputs: `clients` row, `grants` table
- Scoring: `score_grant_match(client_id, grant_id)` function (already in DB, migration 001)
- Threshold: upsert `grant_matches` only where score > 35
- Claude output: match_reasons for top 10 matches only (never rates/APR/loan terms)
- Email: logs to `agent_emails` BEFORE sending; `requires_client_approval = false`
- Compliance log: every sent email logged to `compliance_log` with `contained_loan_terms = false`

## Security Requirements

- S10-002: All client data wrapped with `sanitizeForPrompt()` before Claude calls
- S14-002: All Claude output scanned with `scanOutputConfidentiality()` — if loan terms detected, Telegram alert + abort (never store)
- ABSOLUTE ZERO: Never include interest rates, APR, points, or loan terms in any output
- Circuit breaker on Claude API calls (use `CircuitBreaker` from `agents/lib/circuit-breaker.js`)
- Zod validation on all client data before use
- Deduplicate: upsert `grant_matches` with `onConflict: 'client_id,grant_id'` — no duplicate records
- Verify client `subscription_status = 'active'` and `onboarding_completed = true` before running

### Task 2: agents/capital-grant-finder/score.js

```javascript
'use strict'
const { getServiceClient } = require('../lib/supabase')
const logger = require('../lib/logger')

const AGENT_NAME = 'capital-grant-finder'
const SCORE_THRESHOLD = 35

async function scoreGrantsForClient(clientId) {
  const supabase = getServiceClient()

  // Get all active grants
  const { data: grants, error: grantErr } = await supabase
    .from('grants')
    .select('id, name, grantor, max_award_amount, available_to_tiers')
    .eq('is_active', true)

  if (grantErr) throw new Error(`Failed to fetch grants: ${grantErr.message}`)
  if (!grants || grants.length === 0) return []

  // Get client subscription tier for tier filtering
  const { data: client, error: clientErr } = await supabase
    .from('clients')
    .select('subscription_tier')
    .eq('id', clientId)
    .single()

  if (clientErr) throw new Error(`Client not found: ${clientErr.message}`)

  const scoredMatches = []

  for (const grant of grants) {
    // Skip if grant not available to client's tier
    if (grant.available_to_tiers && !grant.available_to_tiers.includes(client.subscription_tier)) {
      continue
    }

    // Use DB function for scoring (keeps scoring logic in DB, not agent)
    const { data: scoreResult, error: scoreErr } = await supabase
      .rpc('score_grant_match', { p_client_id: clientId, p_grant_id: grant.id })

    if (scoreErr) {
      logger.warn(AGENT_NAME, 'Score error', { grantId: grant.id, error: scoreErr.message })
      continue
    }

    const score = scoreResult || 0
    if (score >= SCORE_THRESHOLD) {
      scoredMatches.push({ grant, score })
    }
  }

  // Sort by score desc
  return scoredMatches.sort((a, b) => b.score - a.score)
}

module.exports = { scoreGrantsForClient }
```

### Task 3: agents/capital-grant-finder/match-reasons.js

```javascript
'use strict'
const { createMessage } = require('../lib/anthropic')
const { sanitizeForPrompt, scanOutputConfidentiality } = require('../lib/sanitize')
const { CircuitBreaker } = require('../lib/circuit-breaker')
const logger = require('../lib/logger')

const AGENT_NAME = 'capital-grant-finder'
const breaker = new CircuitBreaker({ threshold: 3, resetTimeoutMs: 120000 })

async function generateMatchReasons(client, topMatches) {
  // Only generate reasons for top 10 matches
  const top10 = topMatches.slice(0, 10)

  const matchList = top10.map((m, i) =>
    `${i + 1}. ${m.grant.name} (${m.grant.grantor}) — Score: ${m.score} — Max: $${m.grant.max_award_amount?.toLocaleString() || 'varies'}`
  ).join('\n')

  // All client data wrapped in XML delimiters (S10-002)
  const safeBusinessName = sanitizeForPrompt(client.business_name)
  const safeIndustry = sanitizeForPrompt(client.industry)
  const safeState = sanitizeForPrompt(client.state)
  const safeBusinessType = sanitizeForPrompt(client.business_type)
  const safeFlags = sanitizeForPrompt([
    client.is_minority_owned && 'minority-owned',
    client.is_woman_owned && 'woman-owned',
    client.is_veteran_owned && 'veteran-owned',
    client.is_nonprofit && 'nonprofit',
    client.is_rural_business && 'rural',
  ].filter(Boolean).join(', ') || 'none')

  const systemPrompt = `You are a grant research specialist helping small businesses find funding. 
Your role is STRICTLY EDUCATIONAL. You explain grant eligibility and match quality.
CRITICAL RULES — violation results in immediate termination of this agent:
1. NEVER mention interest rates, APR, loan terms, points, or any loan pricing in any form
2. NEVER make guarantees about award likelihood
3. Use professional, encouraging, factual language only
4. Each reason must be 1-2 sentences maximum
5. Focus on eligibility fit, not financial projections`

  const userPrompt = `Client profile:
Business name: ${safeBusinessName}
Industry: ${safeIndustry}
State: ${safeState}
Business type: ${safeBusinessType}
Special qualifications: ${safeFlags}

Top grant matches:
${matchList}

For each grant, write one brief reason why this client is a strong match. Focus on eligibility criteria (location, business type, size, special status). Keep each reason under 2 sentences. Return as JSON array: [{"grant_name": "...", "reason": "..."}]`

  const response = await breaker.call(() => createMessage({
    systemPrompt,
    userPrompt,
    maxTokens: 1024,
  }))

  const rawText = response.content[0]?.text || '[]'

  // CRITICAL: Scan for loan terms before using output (S14-002)
  const scan = scanOutputConfidentiality(rawText)
  if (!scan.safe) {
    logger.error(AGENT_NAME, 'LOAN TERMS DETECTED in Claude output — aborting', { flags: scan.flags })
    throw new Error('Output failed confidentiality scan — loan terms detected')
  }

  try {
    const jsonMatch = rawText.match(/\[[\s\S]*\]/)
    return jsonMatch ? JSON.parse(jsonMatch[0]) : []
  } catch {
    logger.warn(AGENT_NAME, 'Failed to parse match reasons JSON', { raw: rawText.slice(0, 200) })
    return []
  }
}

module.exports = { generateMatchReasons }
```

const { logAndSend } = require('../lib/email')
const { getServiceClient } = require('../lib/supabase')
const logger = require('../lib/logger')

  const supabase = getServiceClient()

  const topGrantLines = topMatches.slice(0, 5).map((m) =>
    `<li><strong>${m.grant.name}</strong> — Up to $${m.grant.max_award_amount?.toLocaleString() || 'varies'} from ${m.grant.grantor}</li>`
  ).join('\n')

  const bodyHtml = `
<p>Hi ${client.owner_name},</p>
<p>Great news! Cosby Capital's AI has identified <strong>${matchCount} grant opportunities</strong> for ${client.business_name}.</p>
<p>Here are your top matches:</p>
<ul>
${topGrantLines}
</ul>
<p>Log in to your dashboard to view all matches, eligibility details, and next steps.</p>
<p><a href="https://cosbycapital.com/dashboard/grants" style="background:#0A1628;color:#C9A84C;padding:12px 24px;text-decoration:none;border-radius:4px;display:inline-block;margin:16px 0">View My Grant Matches →</a></p>
<p><em>Cosby Capital provides grant research and advisory services. We do not guarantee award outcomes. 
All grant applications are submitted by the client or on the client's behalf with explicit written authorization.</em></p>`

  await logAndSend(supabase, {
    agentName: AGENT_NAME,
    clientId: client.id,
    emailType: 'grant_matches_ready',
    toEmail: client.email,
    toName: client.owner_name,
    subject: `Your grant matches are ready — ${matchCount} opportunities found`,
    bodyHtml,
    requiresClientApproval: false,
  })

  // Compliance log — confirm no loan terms
  await supabase.from('compliance_log').insert({
    agent_name: AGENT_NAME,
    client_id: client.id,
    communication_type: 'grant_match_email',
    recipient_email: client.email,
    subject: `Grant matches ready — ${matchCount} opportunities`,
    content_summary: `Notified client of ${matchCount} grant matches`,
    contained_loan_terms: false,
  }).catch((e) => logger.warn(AGENT_NAME, 'Compliance log failed', { error: e.message }))
}

module.exports = { sendGrantMatchEmail }

### Task 5: agents/capital-grant-finder/index.js (main entry)

```javascript
'use strict'
require('dotenv').config()
const cron = require('node-cron')
const { z } = require('zod')
const { getServiceClient } = require('../lib/supabase')
const { sendAlert } = require('../lib/telegram')
const logger = require('../lib/logger')
const { scoreGrantsForClient } = require('./score')
const { generateMatchReasons } = require('./match-reasons')
const { sendGrantMatchEmail } = require('./email')

const AGENT_NAME = 'capital-grant-finder'

const ClientSchema = z.object({
  id: z.string().uuid(),
  business_name: z.string().min(1),
  owner_name: z.string().min(1),
  email: z.string().email(),
  industry: z.string().nullable(),
  state: z.string().nullable(),
  business_type: z.string().nullable(),
  subscription_tier: z.string(),
  subscription_status: z.string(),
  onboarding_completed: z.boolean(),
  is_minority_owned: z.boolean(),
  is_woman_owned: z.boolean(),
  is_veteran_owned: z.boolean(),
  is_nonprofit: z.boolean(),
  is_rural_business: z.boolean(),
})

async function runForClient(clientId, triggerType = 'scheduled') {
  const supabase = getServiceClient()
  const startMs = Date.now()

  // Fetch and validate client
  const { data: rawClient, error: clientErr } = await supabase
    .from('clients')
    .select('*')
    .eq('id', clientId)
    .single()

  if (clientErr || !rawClient) throw new Error(`Client ${clientId} not found`)

  const client = ClientSchema.parse(rawClient)

  // Skip if not active or onboarding incomplete
  if (client.subscription_status !== 'active' || !client.onboarding_completed) {
    logger.info(AGENT_NAME, 'Skipping — client not active/complete', { clientId })
    return
  }

  logger.info(AGENT_NAME, 'Running for client', { clientId, triggerType })

  // Score all grants
  const scoredMatches = await scoreGrantsForClient(clientId)

  if (scoredMatches.length === 0) {
    logger.info(AGENT_NAME, 'No matches above threshold', { clientId })
    await supabase.from('agent_logs').insert({
      agent_name: AGENT_NAME,
      client_id: clientId,
      action: 'grant_finder_run',
      result: 'no_matches',
      duration_ms: Date.now() - startMs,
    })
    return
  }

  // Upsert grant_matches
  const upserts = scoredMatches.map((m) => ({
    client_id: clientId,
    grant_id: m.grant.id,
    match_score: m.score,
    status: 'new',
  }))

  await supabase.from('grant_matches')
    .upsert(upserts, { onConflict: 'client_id,grant_id', ignoreDuplicates: false })

  // Generate Claude reasons for top 10
  let reasonsMap = {}
  try {
    const reasons = await generateMatchReasons(client, scoredMatches)
    for (const r of reasons) {
      const match = scoredMatches.find((m) => m.grant.name === r.grant_name)
      if (match) reasonsMap[match.grant.id] = r.reason
    }

    // Update match_reasons in DB
    for (const [grantId, reason] of Object.entries(reasonsMap)) {
      await supabase.from('grant_matches')
        .update({ match_reasons: [reason] })
        .eq('client_id', clientId)
        .eq('grant_id', grantId)
    }
  } catch (err) {
    logger.warn(AGENT_NAME, 'Match reasons generation failed', { clientId, error: err.message })
    // Continue — matches are saved, just without reasons
  }

  // Email the client
  await sendGrantMatchEmail(client, scoredMatches.length, scoredMatches)

  await supabase.from('agent_logs').insert({
    agent_name: AGENT_NAME,
    client_id: clientId,
    action: 'grant_finder_run',
    result: JSON.stringify({ matches: scoredMatches.length, trigger: triggerType }),
    duration_ms: Date.now() - startMs,
  })

  logger.info(AGENT_NAME, 'Run complete', { clientId, matches: scoredMatches.length })
}

async function runAllClients() {
  const supabase = getServiceClient()
  logger.info(AGENT_NAME, 'Weekly run started')

  const { data: clients, error } = await supabase
    .from('clients')
    .select('id')
    .eq('subscription_status', 'active')
    .eq('onboarding_completed', true)

  if (error) throw new Error(`Failed to fetch clients: ${error.message}`)

  logger.info(AGENT_NAME, `Processing ${clients?.length || 0} clients`)

  for (const { id } of clients || []) {
    await runForClient(id, 'scheduled').catch((err) => {
      logger.error(AGENT_NAME, 'Client run failed', { clientId: id, error: err.message })
    })
  }
}

async function processTriggerQueue() {
  const supabase = getServiceClient()

  const { data: triggers, error } = await supabase
    .from('agent_trigger_queue')
    .select('*')
    .eq('agent_name', AGENT_NAME)
    .eq('status', 'pending')
    .order('created_at', { ascending: true })
    .limit(10)

  if (error || !triggers?.length) return

  for (const trigger of triggers) {
    try {
      await supabase.from('agent_trigger_queue')
        .update({ status: 'processing' })
        .eq('id', trigger.id)

      await runForClient(trigger.client_id, trigger.trigger_type)

      await supabase.from('agent_trigger_queue')
        .update({ status: 'done', processed_at: new Date().toISOString() })
        .eq('id', trigger.id)
    } catch (err) {
      logger.error(AGENT_NAME, 'Trigger queue error', { triggerId: trigger.id, error: err.message })
      const retries = (trigger.retry_count || 0) + 1
      await supabase.from('agent_trigger_queue').update({
        status: retries >= 3 ? 'failed' : 'pending',
        retry_count: retries,
        error: err.message,
      }).eq('id', trigger.id)
    }
  }
}

// Monday 7am UTC
cron.schedule('0 7 * * 1', runAllClients)

// Poll trigger queue every 5 minutes
cron.schedule('*/5 * * * *', processTriggerQueue)

// Process queue on startup (catch any missed triggers)
processTriggerQueue().catch((e) => logger.error(AGENT_NAME, 'Startup trigger poll failed', { error: e.message }))

logger.info(AGENT_NAME, 'capital-grant-finder started', { schedule: 'Monday 7am UTC + trigger queue every 5min' })
```
