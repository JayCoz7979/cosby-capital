# Plan 03-01: Shared Agent Utilities + Agent Infrastructure Migration

## Goal

Create the `agents/` Node.js workspace with shared utilities used by every agent, plus Supabase migration 000010 for tables that agents need (`agent_emails`, `compliance_log`, `platform_health_log`, `agent_trigger_queue`). This plan must complete before any agent plan (03-02 through 03-05).

## Context

All 18 agents run as Node.js services on Railway. They:
- Have NO public HTTP domain (Settings → Networking → no public domain)
- Communicate exclusively through Supabase (service_role key)
- Receive on-signup triggers via `agent_trigger_queue` table (polled every 5 min)
- Run on cron schedules via `node-cron`
- Use pinned model `claude-sonnet-4-6` (never 'latest') — enforced in `agents/lib/anthropic.js`
- Log every action to `agent_logs`
- Log every client-facing communication to `compliance_log`

## Security Requirements

- S10-002: `sanitizeForPrompt()` wraps ALL user data in `<user_content>...</user_content>` XML delimiters
- S14-002: `scanOutputConfidentiality()` scans all Claude outputs for PII + loan terms before storage
- Zero tolerance: Never output interest rates, APR, points, or loan terms in any form
- S10-001: `sanitizeForLog()` redacts EIN, SSN, credit scores, financial data from all logs
- `agents/lib/anthropic.js` verifies `ANTHROPIC_API_URL` is the expected Anthropic endpoint
- `agents/lib/telegram.js` signs every alert with HMAC-SHA256 using `TELEGRAM_SIGNING_SECRET`
- `INTERNAL_SERVICE_SECRET` header check on any inter-service call (32+ byte hex, generated with openssl rand -hex 32)
- `agents/lib/supabase.js` uses `SUPABASE_SERVICE_KEY` (never anon key) with `'server-only'` equivalent pattern

### Task 2: agents/lib/sanitize.js

```javascript
'use strict'
// S10-002: wraps ALL user-supplied data in XML delimiters before including in prompts
// This prevents prompt injection by making user content structurally distinct

const LOAN_TERM_PATTERNS = [
  /\b\d+(\.\d+)?%\s*(APR|interest|annual|rate)/i,
  /\b(APR|annual percentage rate|interest rate)\b/i,
  /\bpoints?\b/i,
  /\b\d+(\.\d+)?\s*points?\s*(origination|discount)/i,
]

function sanitizeForPrompt(value) {
  if (value === null || value === undefined) return '<user_content></user_content>'
  const str = String(value)
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/\\/g, '\\\\')
  return `<user_content>${str}</user_content>`
}

function scanOutputConfidentiality(text) {
  if (!text || typeof text !== 'string') return { safe: true, flags: [] }
  const flags = []

  // Check for loan terms (zero tolerance)
  for (const pattern of LOAN_TERM_PATTERNS) {
    if (pattern.test(text)) {
      flags.push({ type: 'loan_term', pattern: pattern.toString(), matched: text.match(pattern)?.[0] })
    }
  }

  // Check for SSN patterns
  if (/\b\d{3}-\d{2}-\d{4}\b/.test(text)) flags.push({ type: 'ssn' })

  // Check for EIN patterns  
  if (/\b\d{2}-\d{7}\b/.test(text)) flags.push({ type: 'ein' })

  // Check for credit score leak
  if (/\bcredit score[:\s]+\d{3}\b/i.test(text)) flags.push({ type: 'credit_score' })

  return { safe: flags.length === 0, flags }
}

module.exports = { sanitizeForPrompt, scanOutputConfidentiality }
```

### Task 3: agents/lib/logger.js

```javascript
'use strict'
// S10-001: redact PII from all log output

const REDACT_PATTERNS = [
  { pattern: /\b\d{3}-\d{2}-\d{4}\b/g, label: '[SSN_REDACTED]' },
  { pattern: /\b\d{2}-\d{7}\b/g, label: '[EIN_REDACTED]' },
  { pattern: /\b\d{13,19}\b/g, label: '[CARD_REDACTED]' },
  { pattern: /sk-ant-api[^\s]*/g, label: '[API_KEY_REDACTED]' },
  { pattern: /eyJ[a-zA-Z0-9+/=]{40,}/g, label: '[JWT_REDACTED]' },
]

function sanitizeForLog(value) {
  if (!value) return value
  let str = typeof value === 'string' ? value : JSON.stringify(value)
  for (const { pattern, label } of REDACT_PATTERNS) {
    str = str.replace(pattern, label)
  }
  return str
}

function log(agentName, level, message, meta = {}) {
  const entry = {
    ts: new Date().toISOString(),
    agent: agentName,
    level,
    msg: sanitizeForLog(message),
    ...Object.fromEntries(
      Object.entries(meta).map(([k, v]) => [k, sanitizeForLog(v)])
    ),
  }
  // Railway logs stdout as structured JSON
  console.log(JSON.stringify(entry))
}

module.exports = {
  sanitizeForLog,
  info: (agent, msg, meta) => log(agent, 'info', msg, meta),
  warn: (agent, msg, meta) => log(agent, 'warn', msg, meta),
  error: (agent, msg, meta) => log(agent, 'error', msg, meta),
}
```

### Task 4: agents/lib/circuit-breaker.js

```javascript
'use strict'
// Tracks Claude API failures and opens circuit to prevent cascade

class CircuitBreaker {
  constructor({ threshold = 3, resetTimeoutMs = 60000 } = {}) {
    this.threshold = threshold
    this.resetTimeoutMs = resetTimeoutMs
    this.failures = 0
    this.state = 'closed' // closed | open | half-open
    this.openedAt = null
  }

  async call(fn) {
    if (this.state === 'open') {
      const elapsed = Date.now() - this.openedAt
      if (elapsed > this.resetTimeoutMs) {
        this.state = 'half-open'
      } else {
        throw new Error(`Circuit breaker OPEN — retry in ${Math.ceil((this.resetTimeoutMs - elapsed) / 1000)}s`)
      }
    }
    try {
      const result = await fn()
      if (this.state === 'half-open') {
        this.failures = 0
        this.state = 'closed'
      }
      return result
    } catch (err) {
      this.failures++
      if (this.failures >= this.threshold) {
        this.state = 'open'
        this.openedAt = Date.now()
      }
      throw err
    }
  }

  get isOpen() { return this.state === 'open' }
}

module.exports = { CircuitBreaker }
```

### Task 5: agents/lib/telegram.js

```javascript
'use strict'
const https = require('https')
const crypto = require('crypto')

// HMAC-signed Telegram alerts using TELEGRAM_SIGNING_SECRET
function signPayload(payload) {
  const secret = process.env.TELEGRAM_SIGNING_SECRET
  if (!secret) throw new Error('TELEGRAM_SIGNING_SECRET not set')
  return crypto.createHmac('sha256', secret).update(payload).digest('hex')
}

async function sendAlert(message, level = 'info') {
  const token = process.env.TELEGRAM_BOT_TOKEN
  const chatId = process.env.TELEGRAM_CHAT_ID
  if (!token || !chatId) return // silently skip if not configured

  const emoji = { info: 'ℹ️', warn: '⚠️', error: '🚨', success: '✅' }[level] || 'ℹ️'
  const text = `${emoji} ${message}`
  const payload = JSON.stringify({ chat_id: chatId, text, parse_mode: 'HTML' })
  const sig = signPayload(payload)

  return new Promise((resolve, reject) => {
    const req = https.request(
      `https://api.telegram.org/bot${token}/sendMessage`,
      { method: 'POST', headers: { 'Content-Type': 'application/json', 'X-Signature': sig } },
      (res) => {
        let data = ''
        res.on('data', (chunk) => data += chunk)
        res.on('end', () => resolve(JSON.parse(data)))
      }
    )
    req.on('error', reject)
    req.write(payload)
    req.end()
  })
}

module.exports = { sendAlert }
```

```javascript
'use strict'
const Anthropic = require('@anthropic-ai/sdk')

// Verify we're calling the real Anthropic endpoint (endpoint lock)
const EXPECTED_BASE_URL = 'https://api.anthropic.com'
const PINNED_MODEL = 'claude-sonnet-4-6'

let _client = null

function getClient() {
  if (_client) return _client
  
  const apiKey = process.env.ANTHROPIC_API_KEY
  if (!apiKey) throw new Error('ANTHROPIC_API_KEY not set')

  _client = new Anthropic({ apiKey, baseURL: EXPECTED_BASE_URL, maxRetries: 2, timeout: 60000 })
  return _client
}

// Always use pinned model — never accept 'latest' or arbitrary model strings
async function createMessage({ systemPrompt, userPrompt, maxTokens = 2048 }) {
  const client = getClient()
  return client.messages.create({
    model: PINNED_MODEL,
    max_tokens: maxTokens,
    system: systemPrompt,
    messages: [{ role: 'user', content: userPrompt }],
  })
}

module.exports = { getClient, createMessage, PINNED_MODEL }
```

### Task 8: agents/lib/email.js (Resend wrapper)

```javascript
'use strict'
const { Resend } = require('resend')

let _resend = null

function getResend() {
  if (_resend) return _resend
  const apiKey = process.env.RESEND_API_KEY
  if (!apiKey) throw new Error('RESEND_API_KEY not set')
  _resend = new Resend(apiKey)
  return _resend
}

const EMAIL_FOOTER = `
<br><br>
<hr style="border:0;border-top:1px solid #eee;margin:24px 0">
<p style="font-size:12px;color:#666;margin:0">
  Cosby Capital | cosbycapital.com<br>
  Powered by Cosby AI Solutions | cosbyaisolutions.com<br>
  This email was sent on behalf of your business by Cosby Capital's AI consulting system.
  Cosby Capital is a business consulting platform, not a licensed financial advisor or loan broker.
  Questions? <a href="mailto:support@cosbycapital.com">support@cosbycapital.com</a>
</p>`

// Log to agent_emails BEFORE sending (rule 1 for every agent email)
async function logAndSend(supabase, { agentName, clientId, emailType, toEmail, toName, subject, bodyHtml, requiresClientApproval = false }) {
  // Log first — always
  const { data: logged, error: logErr } = await supabase
    .from('agent_emails')
    .insert({
      agent_name: agentName,
      client_id: clientId,
      email_type: emailType,
      to_email: toEmail,
      to_name: toName,
      subject,
      body_html: bodyHtml,
      requires_client_approval: requiresClientApproval,
      client_approved: !requiresClientApproval, // auto-approved if no approval needed
    })
    .select()
    .single()

  if (logErr) throw new Error(`Failed to log email: ${logErr.message}`)

  // Only send if no approval needed
  if (requiresClientApproval) {
    return { queued: true, emailId: logged.id }
  }

  const resend = getResend()
  const fullHtml = bodyHtml + EMAIL_FOOTER
  
  const { data: sent, error: sendErr } = await resend.emails.send({
    from: 'Cosby Capital <info@cosbycapital.com>',
    to: [toEmail],
    subject,
    html: fullHtml,
  })

  if (sendErr) {
    await supabase.from('agent_emails').update({ error: sendErr.message }).eq('id', logged.id)
    throw new Error(`Email send failed: ${sendErr.message}`)
  }

  await supabase.from('agent_emails').update({
    sent: true,
    sent_at: new Date().toISOString(),
    resend_message_id: sent.id,
  }).eq('id', logged.id)

  return { sent: true, emailId: logged.id, messageId: sent.id }
}

module.exports = { logAndSend, EMAIL_FOOTER }
```
